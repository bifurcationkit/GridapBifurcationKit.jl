We consider the cubic–quintic complex Ginzburg–Landau equation (CGE) on a disk $\Omega$ with homogeneous Neumann boundary conditions:

\[\partial_t A = r A + i\,\delta\,\nu\, A + \Delta A - (c_3 + i\mu)\,|A|^2 A - c_5 |A|^4 A,\]

where $A = u_1 + i u_2$ is a complex field. In addition, the advection term $s\,(-y\partial_x + x\partial_y)$ allows us to work in a comoving frame rotating with angular velocity $s$.

The trivial state $A\equiv 0$ is a solution for all $r$. We first continue it with respect to $r$ and detect Hopf bifurcations. The problem is equivariant under rotations (generated by $\partial_\theta = -y\partial_x + x\partial_y$) and under the phase symmetry $A\mapsto e^{i\phi}A$ (generator $iA$). Using these two symmetries, we then continue travelling waves (rotating solutions) and standing waves (pure oscillations) with the travelling wave functional of BifurcationKit.

Building the disk mesh with Gmsh

We build the geometry of a disk of radius $\pi$ from four circle arcs and generate a 2d mesh:

function buildDisk(h::Float64, R, filename = "MyDisk-CGE.msh")    gmsh.initialize()    model = gmsh.model    # center of the disk    center = model.geo.addPoint(0, 0, 0, h, -1)    # four points on the circle    points = []    Nd = 4    for j in 1:Nd        push!(points, model.geo.addPoint(R*cos(2*pi*j/Nd), R*sin(2*pi*j/Nd), 0, h))    end    # four circle arcs    lines = []    for j in 1:Nd        push!(lines, model.geo.addCircleArc(points[j], center, points[j+1 < Nd+1 ? j + 1 : 1]))    end    # curve loop and surface    curveloop = model.geo.addCurveLoop([1, 2, 3, 4])    disk = model.geo.addPlaneSurface([curveloop])    # synchronize CAD with the GMSH model    gmsh.model.geo.synchronize()    # physical groups    t1 = gmsh.model.addPhysicalGroup(1, lines, 1)    gmsh.model.addPhysicalGroup(2, [disk], 10)    gmsh.model.setPhysicalName(1, t1, "boundary")    # generate and write the mesh    model.mesh.generate(2)    gmsh.write(filename)    gmsh.finalize()end# specify a rough number of triangles NbuildDisk(N::Int, R, filename = "MyDisk-CGE.msh") = buildDisk(√(2*R*pi/N), R, filename)buildDisk(250, pi)

Weak forms

Writing $A = u_1 + i u_2$ and $|A|^2 = u_1^2 + u_2^2$, the system reads

\[\begin{aligned} \partial_t u_1 &= r u_1 - \nu u_2 - |A|^2 (c_3 u_1 - \mu u_2) - c_5 |A|^4 u_1,\\ \partial_t u_2 &= r u_2 + \delta\nu u_1 - |A|^2 (c_3 u_2 + \mu u_1) - c_5 |A|^4 u_2. \end{aligned}\]

The mass matrix m0 makes the problem a (linear) differential-algebraic one, for which BifurcationKit automatically uses a generalized eigensolver. We also provide the weak forms of the two Lie generators, $\partial_\theta$ (rotation) and $i$ (phase):

# advection field of the comoving frame: (-y, x) = ∂θadv(x) = VectorValue(-x[2], x[1])function res((u1,u2), p, (v1,v2))    ua(U1, U2) = U1^2 + U2^2    f1(U1, U2) = p.r*U1 - p.ν*U2 - ua(U1,U2)*(p.c3*U1 - p.μ*U2) - p.c5*ua(U1,U2)^2*U1    f2(U1, U2) = p.r*U2 + (p.δ*p.ν)*U1 - ua(U1,U2)*(p.c3*U2 + p.μ*U1) - p.c5*ua(U1,U2)^2*U2    ∫( ( f1∘(u1,u2))*v1     + ( f2∘(u1,u2))*v2     - ∇(u1)⋅∇(v1) - ∇(u2)⋅∇(v2)     - p.s*((adv⋅∇(u1))*v1 + (adv⋅∇(u2))*v2) )*dΩendfunction jac((u1,u2), p, (du1,du2), (v1,v2))    ua(X1, X2) = X1^2 + X2^2    d1f1(X1, X2) = p.r - 2X1*(p.c3*X1 - p.μ*X2) - p.c3*ua(X1,X2) - p.c5*(ua(X1,X2)^2 + 4X1^2*ua(X1,X2))    d2f1(X1, X2) = -p.ν - 2p.c3*X1*X2 + 2p.μ*X2^2 + p.μ*ua(X1,X2) - 4p.c5*ua(X1,X2)*X1*X2    d1f2(X1, X2) = p.δ*p.ν - 2p.c3*X1*X2 - 2p.μ*X1^2 - p.μ*ua(X1,X2) - 4p.c5*ua(X1,X2)*X1*X2    d2f2(X1, X2) = p.r - 2X2*(p.c3*X2 + p.μ*X1) - p.c3*ua(X1,X2) - p.c5*(ua(X1,X2)^2 + 4X2^2*ua(X1,X2))    ∫( ( (d1f1∘(u1,u2))*du1*v1 + (d2f1∘(u1,u2))*du2*v1       - ∇(du1)⋅∇(v1)       + (d1f2∘(u1,u2))*du1*v2 + (d2f2∘(u1,u2))*du2*v2       - ∇(du2)⋅∇(v2)       - p.s*((adv⋅∇(du1))*v1 + (adv⋅∇(du2))*v2) ) )*dΩend# mass matrix (identity on (u1, u2), nonsingular)m0((u1,u2), (v1,v2)) = ∫(u1*v1 + u2*v2)*dΩ# Lie generators: rotation ∂θ and phase iAgenerateur1((u1,u2), p, (du1,du2), (v1,v2)) = ∫( adv⋅∇(du1)*v1 + adv⋅∇(du2)*v2) * dΩgenerateur2((u1,u2), p, (du1,du2), (v1,v2)) = ∫( -du2*v1 + du1*v2 )*dΩ# L2-normalised norm of |A|function N2(x)    u1 = zero(X)    u1.free_values .= x    n = sum(∫(u1[1]*u1[1]+u1[2]*u1[2])dΩ)    s = sum(∫(1)dΩ)    sqrt(n/s)end

Travelling waves aka spiral waves

Because of the rotation equivariance, the Hopf bifurcation points on the trivial branch are also bifurcations to travelling waves, i.e. solutions of the form $A(x,t) = e^{i\omega t} U(x - s\,t)$. Equivalently, these are equilibria $(U, s)$ of the problem in the comoving frame with the phase condition $\langle U - U_0, \partial_\theta U_0\rangle = 0$ and the wave speed $s$ as an additional unknown. This is what TWModel implements, given the generator Gθ. We build an initial guess for the wave from the Hopf eigenvectors with guessFromHopfO2:

function guessFromHopfO2(branch, ind_hopf, eigsolver, M, A, B = 0.; G = (I, I), phase = 0, k = 1.)    specialpoint = branch.specialpoint[ind_hopf]    # parameter value at the Hopf point    p = specialpoint.param    ind_ev = specialpoint.ind_ev    eigenvecs = branch.eig[specialpoint.idx].eigenvecs    # frequency at the Hopf point    ωH = imag(branch.eig[specialpoint.idx].eigenvals[ind_ev]) |> abs    # eigenvectors for the eigenvalues iω    ξ0  = geteigenvector(eigsolver, eigenvecs, ind_ev) |> real    ξ0b = geteigenvector(eigsolver, eigenvecs, ind_ev) |> imag    ξ1  = geteigenvector(eigsolver, eigenvecs, ind_ev - 2) |> real    ξ1b = geteigenvector(eigsolver, eigenvecs, ind_ev - 2) |> imag    basis0 = (ξ0, ξ0b, ξ1, ξ1b)    m = 1    ζ0 = basis0[1]    ζ0b =  G[1] * ζ0 ./ m    ζ1 = basis0[3]    ζ1b = -G[1] * ζ1 ./ m    ζ0 = ζ0 + im .* ζ0b    ζ1 = ζ1 + im .* ζ1b    ζ1 = G[2] * (ζ0)    ζ0 ./=  norm(ζ0)    ζ1 ./=  norm(ζ1)    orbitguess = [real.(specialpoint.x .+                A .* ζ0 .* cis(t - phase) .+                B .* ζ1 .* cis(t - phase)) for t in LinRange(0,2pi,M+1)[1:end-1]]    return (; p, period = 2pi/ωH, guess = orbitguess, x0 = specialpoint.x, ζ0, ζ1, ω = ωH)end

We freeze the travelling wave with a Newton solver, using the Hopf frequency ωH as initial guess for the wave speed $-s$:

Standing waves

Using the phase symmetry $A \mapsto e^{i\phi}A$ instead of the rotation, the same travelling wave functional computes standing waves $A(x,t) = e^{i\omega t}U(x)$:

pred = guessFromHopfO2(br, 2, optn.eigsolver, M, 2*(1+0im), 0*(1+im); k = 1, G = (Gθ, Gphase))
u_ref = copy(pred.guess[1][1:length(prob.u0)])

probSW = BK.TWModel(BK.re_make(prob, params = (par_cge..., r = pred.p - 0.01)),
            Gphase,
            deepcopy(u_ref);
            DAE = [0],
            update_section_every_step = 1,
            jacobian = BK.FullLU())

The standing wave is frozen with a Newton solver:

optn = BK.NewtonPar(max_iterations = 40, tol = 1e-10)
wave_sw = BK.newton(probSW, vcat(u_ref, pred.ω), optn, normN = norminf)
println("ω = ", wave_sw.u[end], ", ||u||₂ = ", N2(wave_sw.u[1:end-probSW.nc]))
plotgridap(wave_sw.u[1:end-probSW.nc])

and continued as a function of $r$, starting backward from the Hopf point:

opt_cont_br = BK.ContinuationPar(p_min = -1.5,
